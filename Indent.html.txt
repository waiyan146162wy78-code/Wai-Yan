    function initSupabase(url, key) {
        try {
            if (window.supabase) {
                supabaseClient = window.supabase.createClient(url, key);
                supabaseClient.channel('public:flowpos_sync')
                    .on('postgres_changes', { event: '*', schema: 'public', table: 'flowpos_data' }, payload => {
                        fetchDataFromCloud();
                    })
                    .subscribe();
            }
        } catch(e) {
            console.error("Supabase init error:", e);
        }
    }

    function saveSupabaseConfig() {
        let url = document.getElementById('supabaseUrlInput').value.trim();
        let key = document.getElementById('supabaseKeyInput').value.trim();
        if(!url || !key) { alert("Supabase URL နှင့် Key ထည့်သွင်းပါ။"); return; }
        localStorage.setItem('flowpos_supabase_url', url);
        localStorage.setItem('flowpos_supabase_key', key);
        savedSupabaseUrl = url;
        savedSupabaseKey = key;
        initSupabase(url, key);
        alert("Supabase ချိတ်ဆက်မှု အချက်အလက် သိမ်းဆည်းပြီးပါပြီ။");
    }

    async function syncDataToCloud() {
        if (!supabaseClient) { alert("ကျေးဇူးပြု၍ ဆက်တင်များတွင် Supabase URL နှင့် Key အရင်ချိတ်ပါ။"); return; }
        let payloadData = {
            shop_name: shopNameGlobal,
            shop_phone: shopPhoneGlobal,
            items: currentShopItems,
            purchases: purchaseList,
            debts: debtList,
            employees: employeeList,
            sales_history: salesHistoryList,
            total_sales: totalSalesAmount,
            total_purchases: totalPurchasesAmount,
            updated_at: new Date()
        };
        try {
            let { error } = await supabaseClient.from('flowpos_data').upsert({ id: 1, data: payloadData });
            if (error) {
                alert("Sync အမှား: " + error.message);
            } else {
                alert("Online သို့ အောင်မြင်စွာ Sync တင်ပြီးပါပြီ။ တခြား User များပါ တခါတည်း ပေါ်လာပါမည်။");
            }
        } catch(err) {
            alert("Cloud Sync Error: " + err.message);
        }
    }

    async function fetchDataFromCloud() {
        if (!supabaseClient) return;
        try {
            let { data, error } = await supabaseClient.from('flowpos_data').select('*').eq('id', 1).single();
            if (data && data.data) {
                let d = data.data;
                if(d.items) currentShopItems = d.items;
                if(d.purchases) purchaseList = d.purchases;
                if(d.debts) debtList = d.debts;
                if(d.employees) employeeList = d.employees;
                if(d.sales_history) salesHistoryList = d.sales_history;
                if(d.total_sales) totalSalesAmount = d.total_sales;
                if(d.total_purchases) totalPurchasesAmount = d.total_purchases;
                renderProductsGrid();
                renderProductManagementTable();
                renderPurchases();
                renderDebts();
                renderEmployees();
                updateProfitLoss();
            }
        } catch(e) {
            console.error("Fetch cloud data error:", e);
        }
    }

    function generateAndSaveActivationCode() {
        let code = document.getElementById('newActivationCodeInput').value.trim().toUpperCase();
        let type = document.getElementById('newActivationTypeInput').value;
        if(!code) { alert("Activation Code ထည့်ပါ။"); return; }
        
        let exists = managedActivationCodes.some(c => c.code === code);
        if(!exists) {
            managedActivationCodes.push({ code, type });
            localStorage.setItem('flowpos_managed_codes', JSON.stringify(managedActivationCodes));
            alert("Activation Code အသစ် ဖန်တီးပြီးပါပြီ။");
            document.getElementById('newActivationCodeInput').value = "";
            renderGeneratedCodesTable();
        } else {
            alert("ဤ Code မှာ ရှိနှင့်ပြီးသား ဖြစ်ပါသည်။");
        }
    }

    function deleteActivationCode(codeVal) {
        if(confirm(`"${codeVal}" ကို ဖျက်ရန် သေချာပါသလား?`)) {
            managedActivationCodes = managedActivationCodes.filter(c => c.code !== codeVal);
            localStorage.setItem('flowpos_managed_codes', JSON.stringify(managedActivationCodes));
            renderGeneratedCodesTable();
        }
    }

    function renderGeneratedCodesTable() {
        let tbody = document.getElementById('generatedCodesTableBody');
        if(!tbody) return;
        tbody.innerHTML = "";
        managedActivationCodes.forEach(item => {
            let badgeClass = item.type === 'creator' ? 'bg-rose-500/20 text-rose-400 border-rose-500/30' : 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30';
            let typeLabel = item.type === 'creator' ? 'ဖန်တီးသူ (Creator)' : 'ဝယ်ယူသူ (Buyer)';
            tbody.innerHTML += `
                <tr class="border-b border-themeBorder">
                    <td class="p-2.5 font-mono font-bold text-themeAccent">${item.code}</td>
                    <td class="p-2.5"><span class="px-2 py-1 rounded-lg text-[10px] border ${badgeClass}">${typeLabel}</span></td>
                    <td class="p-2.5 text-center">
                        <button onclick="deleteActivationCode('${item.code}')" class="text-red-400 hover:text-red-300 px-2 py-1 rounded bg-themeBg border border-themeBorder"><i class="fa-solid fa-trash"></i></button>
                    </td>
                </tr>
            `;
        });
    }

    function checkLogin() {
        let uName = document.getElementById('loginUser').value.trim();
        let uPass = document.getElementById('loginPass').value.trim();
        let isValid = false;
        let isAdminLogin = false;

        if (uName === "admin" && uPass === "12345") {
            isValid = true;
            isAdminLogin = true;
            loggedInUserName = "Admin";
        } else {
            let matchedEmp = employeeList.find(e => e.name === uName && e.code === uPass);
            if (matchedEmp) {
                isValid = true;
                loggedInUserName = matchedEmp.name + " (" + matchedEmp.role + ")";
            }
        }

        if (isValid) {
            // Creator section visibility restricted strictly to Creator Activation Code AND Admin login
            let creatorSection = document.getElementById('creatorOnlySettingsSection');
            if (isCreatorSession && isAdminLogin) {
                creatorSection.classList.remove('hidden');
            } else {
                creatorSection.classList.add('hidden');
            }

            let shopKey = document.getElementById('shopTypeSelect').value;
            currentShopItems = JSON.parse(JSON.stringify(shopDatabase[shopKey].items));
            shopNameGlobal = shopDatabase[shopKey].name;

            document.getElementById('sidebarShopName').innerText = shopNameGlobal;
            document.getElementById('currentLoginUserLabel').innerText = loggedInUserName;
            document.getElementById('headerShopBadge').innerText = shopNameGlobal;
            document.getElementById('voucherShopName').innerText = shopNameGlobal;
            document.getElementById('voucherShopPhone').innerText = shopPhoneGlobal;
            document.getElementById('productSectionTitle').innerText = shopNameGlobal + (currentLanguage === 'my' ? " - ရွေးချယ်ရန်" : " - Items");
            document.getElementById('settingShopName').value = shopNameGlobal;

            renderPOSMainArea(shopKey);
            renderProductManagementTable();
            renderPurchases();
            renderDebts();
            renderEmployees();
            updateProfitLoss();

            document.getElementById('loginScreen').classList.add('hidden');
            document.getElementById('mainDashboard').classList.remove('hidden');
            
            fetchDataFromCloud();
        } else {
            document.getElementById('loginError').classList.remove('hidden');
        }
    }

    function addCustomMoneyTransaction() {
        let provider = document.getElementById('selectedProvider').value;
        let transType = document.getElementById('selectedTransType').value;
        let amount = parseFloat(document.getElementById('customTransAmount').value) || 0;
        let fee = parseFloat(document.getElementById('customTransFee').value);

        if (isNaN(fee)) { alert(currentLanguage === 'my' ? "ဝန်ဆောင်ခ ထည့်သွင်းပါ။" : "Please enter fee."); return; }
        let itemName = `${provider} (${transType})${amount >0 ? ' ['+amount.toLocaleString()+']':''}`;
        addToCart(itemName, fee);
        document.getElementById('customTransAmount').value = "";
        document.getElementById('customTransFee').value = "";
    }

    function renderProductsGrid() {
        let grid = document.getElementById('dynamicProductsGrid');
        if(!grid) return;
        grid.innerHTML = "";
        currentShopItems.forEach((item, index) => {
            let thbPrice = (item.p / exchangeRateTHB).toFixed(2);
            let stockLabel = currentLanguage === 'my' ? 'လက်ကျန်: ' : 'Stock: ';
            let imgHTML = item.img ? `<img src="${item.img}" class="w-full h-20 object-cover rounded-lg mb-2 border border-themeBorder">` : '';
            grid.innerHTML += `
                <div class="bg-themeBg border border-themeBorder p-3.5 rounded-xl transition shadow-sm flex flex-col justify-between">
                    <div onclick="addToCartByName('${item.n}', ${item.p})" class="cursor-pointer">
                        ${imgHTML}
                        <p class="font-medium text-xs truncate">${item.n}</p>
                        <p class="text-[10px] text-themeAccent mt-0.5 font-semibold">${stockLabel}${item.stock}</p>
                    </div>
                    <div class="mt-2 flex items-center justify-between border-t border-themeBorder pt-2">
                        <div>
                            <span class="text-themeAccent font-bold text-xs block">${item.p.toLocaleString()} Ks</span>
                            <span class="text-[10px] opacity-70">${thbPrice} THB</span>
                        </div>
                        <div class="flex items-center gap-1">
                            <button onclick="adjustStock(${index}, -1)" class="w-6 h-6 bg-themeCard border border-themeBorder rounded-lg text-xs flex items-center justify-center hover:bg-themeBorder">-</button>
                            <button onclick="adjustStock(${index}, 1)" class="w-6 h-6 bg-themeCard border border-themeBorder rounded-lg text-xs flex items-center justify-center hover:bg-themeBorder">+</button>
                        </div>
                    </div>
                </div>
            `;
        });
    }

    function adjustStock(index, amount) {
        currentShopItems[index].stock = Math.max(0, currentShopItems[index].stock + amount);
        renderProductsGrid();
        renderProductManagementTable();
    }

    function saveNewProduct() {
        let name = document.getElementById('newProdName').value.trim();
        let price = parseFloat(document.getElementById('newProdPrice').value);
        let stock = parseInt(document.getElementById('newProdStock').value);
        let fileInput = document.getElementById('newProdImg');
        
        if (!name || isNaN(price) || isNaN(stock)) { alert(currentLanguage === 'my' ? "အချက်အလက် အပြည့်အစုံထည့်ပါ။" : "Please fill all details."); return; }

        if (fileInput.files && fileInput.files[0]) {
            let reader = new FileReader();
            reader.onload = function(e) {
                currentShopItems.push({ n: name, p: price, stock: stock, img: e.target.result });
                finalizeNewProductAddition();
            }
            reader.readAsDataURL(fileInput.files[0]);
        } else {
            currentShopItems.push({ n: name, p: price, stock: stock, img: '' });
            finalizeNewProductAddition();
        }
    }

    function switchTab(tabName) {
        const tabs = ['pos', 'products', 'purchases', 'debts', 'profit', 'settings', 'about'];
        let tObj = translations[currentLanguage];
        const titles = {
            'pos': tObj.pageTitlePos,
            'products': tObj.pageTitleProd,
            'purchases': tObj.pageTitlePur,
            'debts': tObj.pageTitleDebt,
            'profit': tObj.pageTitleProf,
            'settings': tObj.pageTitleSet,
            'about': tObj.pageTitleAbt
        };

        tabs.forEach(t => {
            document.getElementById('tab-' + t).classList.add('hidden');
        });

        document.getElementById('tab-' + tabName).classList.remove('hidden');
        document.getElementById('pageTitle').innerText = titles[tabName];
        if(tabName === 'profit') {
            updateProfitLoss();
        }
    }

    function addToCart(name, price) {
        let item = cart.find(i => i.name === name);
        if (item) item.qty += 1;
        else cart.push({ name, price, qty: 1 });
        updateCartUI();
    }

    function addToCartByName(name, price) {
        let prod = currentShopItems.find(p => p.n === name);
        if (prod) {
            if (prod.stock > 0 || prod.stock === 999) {
                if (prod.stock !== 999) {
                    prod.stock -= 1;
                }
                addToCart(name, price);
                renderProductsGrid();
                renderProductManagementTable();
            } else {
                alert(currentLanguage === 'my' ? "စတော့ လက်ကျန် မရှိတော့ပါ။" : "Out of stock.");
            }
        } else {
            addToCart(name, price);
        }
    }

    function removeFromCart(index) { 
        let removedItem = cart[index];
        let prod = currentShopItems.find(p => p.n === removedItem.name);
        if (prod && prod.stock !== 999) {
            prod.stock += removedItem.qty;
        }
        cart.splice(index, 1); 
        updateCartUI();
        renderProductsGrid();
        renderProductManagementTable();
    }

    function clearCart() { 
        cart.forEach(item => {
            let prod = currentShopItems.find(p => p.n === item.name);
            if (prod && prod.stock !== 999) {
                prod.stock += item.qty;
            }
        });
        cart = []; 
        updateCartUI();
        renderProductsGrid();
        renderProductManagementTable();
    }

    function updateCartUI() {
        let tbody = document.getElementById('cartTableBody');
        let grandTotalMMKElem = document.getElementById('grandTotalMMK');
        let grandTotalTHBElem = document.getElementById('grandTotalTHB');
        
        tbody.innerHTML = "";
        if (cart.length === 0) {
            tbody.innerHTML = `<tr><td colspan="4" class="p-4 text-center text-themeMuted">${currentLanguage === 'my' ? 'ပစ္စည်းများ မရွေးရသေးပါ' : 'No items selected'}</td></tr>`;
            grandTotalMMKElem.innerText = "0 Ks";
            grandTotalTHBElem.innerText = "0 THB";
            return;
        }

        let totalMMK = 0;
        cart.forEach((item, index) => {
            let itemTotal = item.price * item.qty;
            totalMMK += itemTotal;
            tbody.innerHTML += `<tr><td class="p-2 font-medium">${item.name}</td><td class="p-2 text-center">${item.qty}</td><td class="p-2 text-right">${formatBothCurrency(itemTotal)}</td><td class="p-2 text-center no-print"><button onclick="removeFromCart(${index})" class="text-red-400"><i class="fa-solid fa-trash"></i></button></td></tr>`;
        });

        grandTotalMMKElem.innerText = totalMMK.toLocaleString() + " Ks";
        grandTotalTHBElem.innerText = (totalMMK / exchangeRateTHB).toFixed(2) + " THB";
    }

    function saveVoucherImage() {
        if (cart.length === 0) { alert(currentLanguage === 'my' ? "ဘောက်ချာတွင် ပစ္စည်းထည့်ပါ။" : "Cart is empty."); return; }

            let sessionTotal = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);

        });
    }
